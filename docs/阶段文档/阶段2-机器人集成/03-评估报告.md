### 3.1 panda_with_water_gun.urdf.xacro

**✓ 正确实现**

```xml
<!-- 亮点 1：正确使用 xacro 参数 -->
<xacro:arg name="arm_id" default="panda"/>

<!-- 亮点 2：初始位置定义在 URDF 中 -->
<xacro:property name="initial_positions" value="${dict(...)}"/>

<!-- 亮点 3：使用宏生成重复的关节配置 -->
<xacro:macro name="add_joint" params="name">
  <joint name="$(arg arm_id)_${name}">
    <state_interface name="position">
      <param name="initial_value">${initial_positions[name]}</param>
    </state_interface>
  </joint>
</xacro:macro>
```
