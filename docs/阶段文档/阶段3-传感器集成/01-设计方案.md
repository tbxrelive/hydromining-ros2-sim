### 任务 2：点云采集节点

**功能**：订阅激光雷达点云，按需保存单帧数据

**节点设计**：
```python
class PointCloudCapture(Node):
    """
    点云采集节点
    
    Subscriptions:
        /hydromining/lidar/points (sensor_msgs/PointCloud2): 原始点云
        /tf, /tf_static: TF 变换
    
    Services:
        /capture_pointcloud (std_srvs/Trigger): 采集并保存当前点云
    
    Parameters:
        output_dir (str): 点云保存目录
        sensor_frame (str): 传感器坐标系（默认 "lidar_link"）
        world_frame (str): 世界坐标系（默认 "world"）
    """
    def __init__(self):
        super().__init__('pointcloud_capture')
        
        # 参数
        self.declare_parameter('output_dir', '~/pointcloud_data')
        self.declare_parameter('sensor_frame', 'lidar_link')
        self.declare_parameter('world_frame', 'world')
        
        # 订阅点云
        qos = QoSProfile(
            reliability=ReliabilityPolicy.BEST_EFFORT,
            depth=1
        )
        self.subscription_ = self.create_subscription(
            PointCloud2,
            '/hydromining/lidar/points',
            self.pointcloud_callback,
            qos
        )
        
        # TF 监听器
        self.tf_buffer_ = Buffer()
        self.tf_listener_ = TransformListener(self.tf_buffer_, self)
        
        # 服务：触发采集
        self.capture_service_ = self.create_service(
            Trigger,
            '/capture_pointcloud',
            self.capture_callback
        )
        
        # 最新点云缓存
        self.latest_pointcloud_ = None
        
        self.get_logger().info('点云采集节点已启动')
    
    def pointcloud_callback(self, msg: PointCloud2):
        """缓存最新点云"""
        self.latest_pointcloud_ = msg
    
    def capture_callback(self, request, response):
        """服务回调：保存当前点云"""
        if self.latest_pointcloud_ is None:
            response.success = False
            response.message = "无可用点云数据"
            return response
        
        try:
            # 1. 获取 TF 变换
            sensor_frame = self.get_parameter('sensor_frame').value
            world_frame = self.get_parameter('world_frame').value
            
            transform = self.tf_buffer_.lookup_transform(
                world_frame,
                sensor_frame,
                rclpy.time.Time()  # 最新可用变换
            )
            
            # 2. 转换点云到世界坐标系
            pointcloud_world = self.transform_pointcloud(
                self.latest_pointcloud_,
                transform
            )
            
            # 3. 保存为 PCD 文件
            timestamp = self.get_clock().now().nanoseconds
            output_dir = os.path.expanduser(
                self.get_parameter('output_dir').value
            )
            os.makedirs(output_dir, exist_ok=True)
            
            filename = f"{output_dir}/pointcloud_{timestamp}.pcd"
            self.save_as_pcd(pointcloud_world, filename)
            
            response.success = True
            response.message = f"点云已保存: {filename}"
            self.get_logger().info(response.message)
            
        except Exception as e:
            response.success = False
            response.message = f"采集失败: {str(e)}"
            self.get_logger().error(response.message)
        
        return response
    
    def transform_pointcloud(self, cloud_msg, transform):
        """将点云从 sensor_frame 转换到 world_frame"""
        # 使用 sensor_msgs_py 或 tf2_sensor_msgs
        from tf2_sensor_msgs.tf2_sensor_msgs import do_transform_cloud
        return do_transform_cloud(cloud_msg, transform)
    
    def save_as_pcd(self, cloud_msg, filename):
        """保存点云为 PCD 格式"""
        import open3d as o3d
        from sensor_msgs_py import point_cloud2
        
        # 转换 ROS 点云 → numpy
        points = point_cloud2.read_points(
            cloud_msg,
            field_names=("x", "y", "z"),
            skip_nans=True
        )
        points_array = np.array(list(points))
        
        # 保存为 PCD
        pcd = o3d.geometry.PointCloud()
        pcd.points = o3d.utility.Vector3dVector(points_array)
        o3d.io.write_point_cloud(filename, pcd)
```

### 任务 3：多视角观测序列

**实现方式**：扩展现有的 `test_motion.py`，新增观测序列

**新脚本**：`scripts/test_observation.py`

```python
#!/usr/bin/env python3
"""
多视角观测测试脚本
按顺序执行：A → B → A，每个位姿采集点云
"""

import rclpy
from rclpy.node import Node
from rclpy.action import ActionClient
from control_msgs.action import FollowJointTrajectory
from trajectory_msgs.msg import JointTrajectoryPoint
from std_srvs.srv import Trigger
import time


class ObservationTest(Node):
    """多视角观测测试节点"""
    
    # 预定义观测位姿（关节角度，单位：弧度）
    VIEWPOINT_A = [0.0, -0.785, 0.0, -2.356, 0.0, 1.571, 0.785]
    VIEWPOINT_B = [0.524, -0.785, 0.0, -2.356, 0.0, 1.571, 0.785]
    
    def __init__(self):
        super().__init__('observation_test')
        
        # Action 客户端：控制机械臂
        self.trajectory_client_ = ActionClient(
            self,
            FollowJointTrajectory,
            '/panda_arm_controller/follow_joint_trajectory'
        )
        
        # Service 客户端：触发点云采集
        self.capture_client_ = self.create_client(
            Trigger,
            '/capture_pointcloud'
        )
        
        self.get_logger().info('等待 Action Server...')
        self.trajectory_client_.wait_for_server()
        
        self.get_logger().info('等待点云采集服务...')
        self.capture_client_.wait_for_service()
        
        self.get_logger().info('✓ 所有服务已就绪')
    
    def move_to_position(self, joint_positions, duration=5.0):
        """移动到指定关节位置"""
        goal_msg = FollowJointTrajectory.Goal()
        
        # 关节名称
        goal_msg.trajectory.joint_names = [
            'panda_joint1', 'panda_joint2', 'panda_joint3', 'panda_joint4',
            'panda_joint5', 'panda_joint6', 'panda_joint7'
        ]
        
        # 目标点
        point = JointTrajectoryPoint()
        point.positions = joint_positions
        point.time_from_start.sec = int(duration)
        goal_msg.trajectory.points.append(point)
        
        self.get_logger().info(f'发送目标位置: {joint_positions}')
        
        # 发送并等待完成
        future = self.trajectory_client_.send_goal_async(goal_msg)
        rclpy.spin_until_future_complete(self, future)
        
        goal_handle = future.result()
        if not goal_handle.accepted:
            self.get_logger().error('目标被拒绝')
            return False
        
        result_future = goal_handle.get_result_async()
        rclpy.spin_until_future_complete(self, result_future)
        
        result = result_future.result().result
        if result.error_code == 0:
            self.get_logger().info('✓ 运动完成')
            return True
        else:
            self.get_logger().error(f'运动失败，错误码: {result.error_code}')
            return False
    
    def capture_pointcloud(self, label):
        """触发点云采集"""
        self.get_logger().info(f'正在采集点云: {label}')
        
        request = Trigger.Request()
        future = self.capture_client_.call_async(request)
        rclpy.spin_until_future_complete(self, future)
        
        response = future.result()
        if response.success:
            self.get_logger().info(f'✓ {response.message}')
        else:
            self.get_logger().error(f'✗ {response.message}')
        
        return response.success
    
    def run_observation_sequence(self):
        """执行完整观测序列"""
        self.get_logger().info('=== 开始多视角观测测试 ===')
        
        # 步骤 1：移动到观测位姿 A
        self.get_logger().info('\n[1/5] 移动到观测位姿 A（正面）')
        if not self.move_to_position(self.VIEWPOINT_A, duration=5.0):
            return False
        
        time.sleep(1.0)  # 等待稳定
        
        # 步骤 2：采集点云 A
        self.get_logger().info('\n[2/5] 采集点云 A')
        if not self.capture_pointcloud('viewpoint_A'):
            return False
        
        # 步骤 3：移动到观测位姿 B
        self.get_logger().info('\n[3/5] 移动到观测位姿 B（侧面 +30°）')
        if not self.move_to_position(self.VIEWPOINT_B, duration=5.0):
            return False
        
        time.sleep(1.0)
        
        # 步骤 4：采集点云 B
        self.get_logger().info('\n[4/5] 采集点云 B')
        if not self.capture_pointcloud('viewpoint_B'):
            return False
        
        # 步骤 5：返回 A，采集基准观测
        self.get_logger().info('\n[5/5] 返回观测位姿 A（基准观测）')
        if not self.move_to_position(self.VIEWPOINT_A, duration=5.0):
            return False
        
        time.sleep(1.0)
        
        self.get_logger().info('\n[5/5] 采集基准观测点云')
        if not self.capture_pointcloud('viewpoint_A_baseline'):
            return False
        
        self.get_logger().info('\n=== ✓ 观测序列完成 ===')
        return True


def main(args=None):
    rclpy.init(args=args)
    node = ObservationTest()
    
    try:
        success = node.run_observation_sequence()
        if success:
            node.get_logger().info('\n✓✓✓ 测试成功！')
        else:
            node.get_logger().error('\n✗✗✗ 测试失败')
    except KeyboardInterrupt:
        node.get_logger().info('用户中断')
    finally:
        node.destroy_node()
        rclpy.shutdown()


if __name__ == '__main__':
    main()
```

### 任务 4：点云配准验证

**工具脚本**：`scripts/verify_registration.py`

```python
#!/usr/bin/env python3
"""
点云配准验证工具
读取保存的点云，执行 ICP 配准，评估对齐质量
"""

import open3d as o3d
import numpy as np
import glob
import os


def load_pointclouds(data_dir):
    """加载所有点云文件"""
    pcd_files = sorted(glob.glob(f"{data_dir}/*.pcd"))
    
    if len(pcd_files) < 2:
        print(f"错误：至少需要 2 个点云文件，但只找到 {len(pcd_files)} 个")
        return []
    
    print(f"找到 {len(pcd_files)} 个点云文件:")
    for f in pcd_files:
        print(f"  - {os.path.basename(f)}")
    
    pointclouds = [o3d.io.read_point_cloud(f) for f in pcd_files]
    return pointclouds, pcd_files


def downsample_and_filter(pcd, voxel_size=0.05):
    """下采样和离群点移除"""
    # 体素下采样
    pcd_down = pcd.voxel_down_sample(voxel_size)
    
    # 统计离群点移除
    pcd_filtered, _ = pcd_down.remove_statistical_outlier(
        nb_neighbors=20,
        std_ratio=2.0
    )
    
    return pcd_filtered


def register_icp(source, target, threshold=0.02):
    """ICP 配准"""
    # 初始对齐（已经在世界坐标系，不需要粗配准）
    trans_init = np.identity(4)
    
    # 点到点 ICP
    reg_result = o3d.pipelines.registration.registration_icp(
        source, target, threshold, trans_init,
        o3d.pipelines.registration.TransformationEstimationPointToPoint(),
        o3d.pipelines.registration.ICPConvergenceCriteria(max_iteration=200)
    )
    
    return reg_result


def evaluate_registration(pcd_a, pcd_b):
    """评估配准质量"""
    print("\n=== 点云配准评估 ===")
    
    # 预处理
    print("预处理点云...")
    pcd_a_down = downsample_and_filter(pcd_a)
    pcd_b_down = downsample_and_filter(pcd_b)
    
    print(f"点云 A: {len(pcd_a_down.points)} 点")
    print(f"点云 B: {len(pcd_b_down.points)} 点")
    
    # 执行 ICP 配准
    print("\n执行 ICP 配准...")
    reg_result = register_icp(pcd_a_down, pcd_b_down, threshold=0.05)
    
    # 输出结果
    print(f"\n配准适应度 (fitness): {reg_result.fitness:.4f}")
    print(f"配准 RMSE: {reg_result.inliers_rmse:.4f} m")
    print(f"对应点对数量: {len(reg_result.correspondence_set)}")
    
    print("\n变换矩阵:")
    print(reg_result.transformation)
    
    # 判断配准质量
    if reg_result.fitness > 0.5 and reg_result.inliers_rmse < 0.05:
        print("\n✓ 配准质量良好（重叠区域对齐成功）")
        verdict = True
    elif reg_result.fitness > 0.3:
        print("\n⚠ 配准质量一般（可能需要调整观测位姿）")
        verdict = False
    else:
        print("\n✗ 配准失败（重叠区域不足或数据质量差）")
        verdict = False
    
    # 可视化
    print("\n显示配准结果（关闭窗口继续）...")
    pcd_a_down.paint_uniform_color([1, 0, 0])  # 红色
    pcd_b_down.paint_uniform_color([0, 1, 0])  # 绿色
    pcd_b_aligned = pcd_b_down.transform(reg_result.transformation)
    
    o3d.visualization.draw_geometries(
        [pcd_a_down, pcd_b_aligned],
        window_name="点云配准结果（红色=A，绿色=B对齐后）"
    )
    
    return verdict


def main():
    import sys
    
    if len(sys.argv) < 2:
        print("用法: python3 verify_registration.py <点云数据目录>")
        sys.exit(1)
    
    data_dir = os.path.expanduser(sys.argv[1])
    
    if not os.path.exists(data_dir):
        print(f"错误：目录不存在: {data_dir}")
        sys.exit(1)
    
    # 加载点云
    pointclouds, filenames = load_pointclouds(data_dir)
    
    if len(pointclouds) < 2:
        sys.exit(1)
    
    # 配准前两个点云（A 和 B）
    print(f"\n对比点云:")
    print(f"  源: {os.path.basename(filenames[0])}")
    print(f"  目标: {os.path.basename(filenames[1])}")
    
    success = evaluate_registration(pointclouds[0], pointclouds[1])
    
    if success:
        print("\n✓✓✓ 验证通过：传感器系统工作正常")
    else:
        print("\n✗✗✗ 验证失败：需要调整配置")


if __name__ == '__main__':
    main()
```

## 七、潜在问题与预案

### 问题 1：点云数据噪声过大

**现象**：采集的点云离群点多，影响配准

**解决**：
1. 在 Gazebo 传感器插件中调整噪声参数
2. 增强 `downsample_and_filter` 的滤波强度

2. 使用 `ros2 run tf2_tools view_frames` 查看 TF 树
3. 在 `lookup_transform` 中添加超时参数
