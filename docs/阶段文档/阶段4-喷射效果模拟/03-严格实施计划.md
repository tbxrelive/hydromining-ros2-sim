```python
class LevelSetEvolver:
    """
    Level Set 演化器
    
    实现:
    ⑪ ∂φ/∂t = ṽ_e * ||∇φ||
    ⑫ φ^(k+1) = φ^k + Δt * ṽ_e^k * ||∇φ^k||
    """

        self.dt = dt

        """计算 ∇φ"""
        # 中央差分或前向差分
        pass
    
    def evolve_step(self, grid: VoxelGrid, 
                   erosion_model: ErosionModel,
                   spray_action) -> VoxelGrid:
        """执行单步Level Set演化"""
        # 1. 计算冲蚀速度 v_e
        v_e = erosion_model.compute_erosion_velocity(grid, spray_action)
        
        # 2. 计算梯度 ∇φ
        grad_phi = self.compute_gradient(grid.phi)
        grad_norm = np.linalg.norm(grad_phi, axis=-1)
        
        # 3. Level Set 演化: φ^(k+1) = φ^k + Δt * v_e * ||∇φ||
        phi_new = grid.phi + self.dt * v_e * grad_norm
        
        # 4. 更新
        grid.update_sdf(phi_new)
        return grid
```

#### 4.3.2 节点实现

**文件**: `scripts/spray_controller_node.py`

```python
#!/usr/bin/env python3
"""
水枪喷射控制节点

订阅:
  - /water_gun_pose (Transform, TF)

发布:
  - /spray_action/feedback (int)
  
服务:
  - /spray_coal (SprayAction)
"""

class SprayControllerNode(Node):
    def __init__(self):
        super().__init__('spray_controller')
        self._logger = self.get_logger()
        
        # 初始化
        self._init_voxel_grid()
        self._init_models()
        self._init_service()
        
        self._logger.info('SprayController ready')
    
    def _init_voxel_grid(self):
        """初始化体素网格"""
        mesh_path = self.declare_parameter(
            'coal_wall_mesh',
            'src/hydromining_sim/models/coal_wall/meshes/coal_wall.obj'
        ).get_parameter_value().string_value
        
        self.voxel_grid = VoxelGrid.from_mesh(
            mesh_path,
            resolution=(128, 128, 64)
        )
    
    def handle_spray_action(self, request, response):
        """处理喷射动作"""
        try:
            # 1. 解析请求
            spray_action = SprayActionData(
                position=np.array([request.position.x, 
                                  request.position.y, 
                                  request.position.z]),
                direction=np.array([request.direction.x, 
                                   request.direction.y, 
                                   request.direction.z]),
                pressure=request.pressure,
                duration=request.duration
            )
            
            # 2. 执行冲蚀计算
            start_time = time.time()
            result = self._apply_spray(spray_action)
            elapsed = time.time() - start_time
            
            # 3. 返回结果
            response.success = True
            response.message = 'Spray applied successfully'
            response.volume_removed = result['volume_removed']
            response.max_depth = result['max_depth']
            response.processing_time_ms = int(elapsed * 1000)
            
        except Exception as e:
            response.success = False
            response.message = f'Error: {str(e)}'
            self._logger.error(f'Spray failed: {e}')
        
        return response
    
    def _apply_spray(self, spray_action):
        """执行喷射"""
        # 1. 创建冲蚀模型和演化器
        erosion_model = ErosionModel(
            R_c_func=DefaultErosionKernel.R_c,
            b_func=DefaultErosionKernel.b
        )
        evolver = LevelSetEvolver(dt=0.01)
        
        # 2. 持续演化（模拟连续喷射）
        volume_removed = 0
        max_depth = 0
        dt_total = spray_action.duration
        steps = int(dt_total / evolver.dt)
        
        for step in range(steps):
            phi_old = self.voxel_grid.phi.copy()
            
            # 更新Level Set
            self.voxel_grid = evolver.evolve_step(
                self.voxel_grid,
                erosion_model,
                spray_action
            )
            
            # 计算统计信息
            vol_step = GeometryUtils.compute_volume_removed(
                phi_old,
                self.voxel_grid.phi,
                spacing=self.voxel_grid.spacing
            )
            volume_removed += vol_step
            
            # 每100步更新Gazebo（1秒刷新）
            if step % 100 == 0 and step > 0:
                self._update_gazebo_model()
        
        # 最终更新
        self._update_gazebo_model()
        
        return {
            'volume_removed': volume_removed,
            'max_depth': max_depth,
            'steps': steps
        }
    
    def _update_gazebo_model(self):
        """更新Gazebo中的煤壁模型"""
        # 1. 转换为Mesh
        vertices, faces = self.voxel_grid.to_mesh()
        
        # 2. 导出.obj
        mesh_path = 'src/hydromining_sim/models/coal_wall/meshes/coal_wall.obj'
        GeometryConverter.marching_cubes_to_obj(self.voxel_grid, mesh_path)
        
        # 3. 通知Gazebo重加载（待实现）
        self._logger.info(f'Updated Mesh: {len(vertices)} vertices')


def main(args=None):
    rclpy.init(args=args)
    node = SprayControllerNode()
    executor = MultiThreadedExecutor(num_threads=2)
    executor.add_node(node)
    executor.spin()
    rclpy.shutdown()


if __name__ == '__main__':
    main()
```
